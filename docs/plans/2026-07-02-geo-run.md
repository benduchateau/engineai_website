### Task 8: FaqSection component + /claude FAQ

Server-rendered accordion using native `<details>/<summary>` (content stays in the DOM = crawlable; zero client JS; no deps). Component emits its own FAQPage JSON-LD from the same data, so visible copy and schema can never drift (a Google requirement).

**Files:**
- Create: `app/components/FaqSection.tsx`
- Modify: `app/claude/page.tsx`

**Interfaces:**
- Consumes: `AnimatedSection` (existing, `delay` prop), `JsonLdScript` (Task 5).
- Produces: `FaqSection({ heading, items, pageUrl, accentClass?, background? })` — default export; `FaqItem = { question: string; answer: string }`. Used by Task 9.

- [ ] **Step 1: Create the component**
```tsx
// app/components/FaqSection.tsx
import AnimatedSection from "./AnimatedSection";
import JsonLdScript from "./JsonLdScript";

export interface FaqItem {
  question: string;
  answer: string;
}

interface FaqSectionProps {
  heading: string;
  items: FaqItem[];
  pageUrl: string;
  accentClass?: string;
  background?: "primary" | "surface";
}

export default function FaqSection({
  heading,
  items,
  pageUrl,
  accentClass = "text-gold",
  background = "primary",
}: FaqSectionProps) {
  const faqSchema = {
    "@context": "https://schema.org",
    "@type": "FAQPage",
    "@id": `${pageUrl}#faq`,
    mainEntity: items.map((item) => ({
      "@type": "Question",
      name: item.question,
      acceptedAnswer: { "@type": "Answer", text: item.answer },
    })),
  };

  return (
    <section
      id="faq"
      className={`relative py-24 md:py-32 px-6 ${
        background === "surface" ? "bg-bg-surface" : "bg-bg-primary"
      }`}
    >
      <JsonLdScript data={faqSchema} />
      <div className="max-w-4xl mx-auto">
        <AnimatedSection>
          <p className="font-mono text-xs tracking-[0.3em] uppercase text-text-secondary mb-6">
            FAQ
          </p>
          <h2 className="text-3xl md:text-4xl font-light leading-[1.15] tracking-tight text-white mb-10">
            {heading}
          </h2>
        </AnimatedSection>
        <div className="max-w-[680px]">
          {items.map((item, i) => (
            <AnimatedSection key={item.question} delay={100 + i * 60}>
              <details className="group border-t border-border py-5">
                <summary className="flex items-start gap-4 cursor-pointer list-none [&::-webkit-details-marker]:hidden">
                  <span className={`font-mono text-xs ${accentClass} mt-1 shrink-0`}>
                    0{i + 1}
                  </span>
                  <span className="flex-1 text-white text-base md:text-lg font-light">
                    {item.question}
                  </span>
                  <span
                    className={`font-mono text-sm ${accentClass} mt-0.5 shrink-0 transition-transform duration-300 group-open:rotate-45`}
                    aria-hidden="true"
                  >
                    +
                  </span>
                </summary>
                <p className="text-text-body leading-[1.8] text-base mt-4 pl-8 max-w-[600px]">
                  {item.answer}
                </p>
              </details>
            </AnimatedSection>
          ))}
          <div className="border-t border-border" />
        </div>
      </div>
    </section>
  );
}
```

```ts
import type { NextConfig } from "next";

const securityHeaders = [
  { key: "X-Frame-Options", value: "DENY" },
  { key: "X-Content-Type-Options", value: "nosniff" },
  { key: "Referrer-Policy", value: "strict-origin-when-cross-origin" },
  { key: "Permissions-Policy", value: "camera=(), microphone=(), geolocation=()" },
  {
    key: "Content-Security-Policy",
    value: [
      "default-src 'self'",
      "script-src 'self' 'unsafe-inline'",
      "style-src 'self' 'unsafe-inline'",
      "img-src 'self' data: blob:",
      "font-src 'self'",
      "connect-src 'self' https://vitals.vercel-insights.com",
      "frame-ancestors 'none'",
      "base-uri 'self'",
      "form-action 'self'",
    ].join("; "),
  },
];

// impeccable live mode injects a helper script from localhost:8400 in dev;
// widen only in development so production CSP stays strict.
const isDev = process.env.NODE_ENV === "development";
const csp = securityHeaders.map((h) =>
  h.key === "Content-Security-Policy" && isDev
    ? {
        ...h,
        value: h.value
          .replace("script-src 'self'", "script-src 'self' http://localhost:8400")
          .replace("connect-src 'self'", "connect-src 'self' http://localhost:8400 ws://localhost:8400"),
      }
    : h
);

const nextConfig: NextConfig = {
  async headers() {
    return [
      {
        source: "/(.*)",
        headers: csp,
      },
    ];
  },
};

export default nextConfig;
```
